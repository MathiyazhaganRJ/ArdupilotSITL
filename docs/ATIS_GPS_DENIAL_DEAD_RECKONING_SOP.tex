\textbf{1. GNSS HARDWARE CUTOFF \& RF JAMMING SIMULATION}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}SIM\_GPS1\_ENABLE} \hfill \textbf{\color{fmsgray}[SITL HARDWARE TOGGLE]} \\
\textbf{DEFAULT:} \texttt{1} (ACTIVE) \hfill \textbf{TARGET:} {\color{fmsamber}\texttt{0} (SEVERED / DENIED)} \\
\textbf{LSK 1L COMMAND:} {\bfseries\color{fmscyan}param set SIM\_GPS1\_ENABLE 0} \\[0.8mm]
\footnotesize Simulates physical antenna severance or receiver power cutoff. SITL instantly sets $fix\_type = 0$ (NO\_FIX) and $satellites\_visible = 0$. EKF3 immediately declares satellite aiding invalid and transitions directly to smooth Dead Reckoning without coordinate jitter.
\end{tcolorbox}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}SIM\_GPS1\_JAM} \hfill \textbf{\color{fmsgray}[ELECTRONIC WARFARE INJECTION]} \\
\textbf{DEFAULT:} \texttt{0} (NORMAL RF) \hfill \textbf{TARGET:} {\color{fmsamber}\texttt{1} (RF JAMMED)} \\
\textbf{LSK 2L COMMAND:} {\bfseries\color{fmscyan}param set SIM\_GPS1\_JAM 1} \\[0.8mm]
\footnotesize Injects broadband RF interference on GPS L1/L2 carrier frequencies ($C/N_0 < 28\text{ dB-Hz}$), inflating pseudorange noise. \textbf{CAUTION:} SITL maintains a degraded 3D fix with jumping coordinates; requires automated gating configuration (Section 2) to eliminate aircraft wandering.
\end{tcolorbox}

\vspace{0.5mm}
\textbf{2. INNOVATION GATING \& ATTITUDE HORIZON PROTECTION}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}EK3\_GLITCH\_RAD} \hfill \textbf{\color{fmsgray}[INNOVATION STEP GATE THRESHOLD]} \\
\textbf{DEFAULT:} \texttt{25.0} m \hfill \textbf{RECOMMENDED:} {\color{fmsgreen}\texttt{15.0} to \texttt{20.0} m} \\
\textbf{MCDU COMMAND:} {\bfseries\color{fmscyan}param set EK3\_GLITCH\_RAD 15} \\[0.8mm]
\footnotesize Defines maximum allowable horizontal step jump between consecutive GPS updates before EKF3 rejects the measurement as anomalous. Rejects abrupt adversarial spoofing coordinates and extreme jamming spikes.
\end{tcolorbox}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}FS\_EKF\_THRESH} \hfill \textbf{\color{fmsgray}[FLIGHT SAFETY FAILSAFE RATIO]} \\
\textbf{DEFAULT:} \texttt{0.80} \hfill \textbf{ANTI-JAM TARGET:} {\color{fmsgreen}\texttt{0.60} to \texttt{0.65}} \\
\textbf{MCDU COMMAND:} {\bfseries\color{fmscyan}param set FS\_EKF\_THRESH 0.60} \\[0.8mm]
\footnotesize Normalized innovation variance ratio that triggers autonomous EKF Failsafe. Lowering to \texttt{0.60} forces ArduPlane to trip failsafe immediately under jamming noise rather than tolerating noisy jumping fixes.
\end{tcolorbox}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}AHRS\_GPS\_USE} \hfill \textbf{\color{fmsgray}[ATTITUDE FILTER ISOLATION]} \\
\textbf{DEFAULT:} \texttt{1} (ENABLED) \hfill \textbf{HOSTILE RF TARGET:} {\color{fmsamber}\texttt{0} (ISOLATE AHRS)} \\
\textbf{MCDU COMMAND:} {\bfseries\color{fmscyan}param set AHRS\_GPS\_USE 0} \\[0.8mm]
\footnotesize Disconnects GPS ground velocity from attitude centrifugal turn compensation ($a_{\text{cent}} = \omega_z \cdot V_{\text{gps}}$). Prevents erratic GPS velocity spikes from corrupting the artificial horizon and inducing violent wing-rocking.
\end{tcolorbox}

\noindent\textbf{3. DEAD RECKONING \& AERODYNAMIC WIND ESTIMATION}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}EK3\_DRAG\_BCOEF\_X / EK3\_DRAG\_BCOEF\_Y} \hfill \textbf{\color{fmsgray}[DRAG BALLISTICS]} \\
\textbf{DEFAULT:} \texttt{0.0} (DISABLED) \hfill \textbf{AIRFRAME TARGET:} {\color{fmsgreen}\texttt{25.0} to \texttt{35.0} ($kg/m^2$)} \\
\textbf{LSK 4L COMMANDS:} {\bfseries\color{fmscyan}param set EK3\_DRAG\_BCOEF\_X 25} \quad|\quad {\bfseries\color{fmscyan}param set EK3\_DRAG\_BCOEF\_Y 25} \\[0.8mm]
\footnotesize Calibrates longitudinal and lateral airframe ballistic drag. In GPS-denied flight, EKF3 combines pitot true airspeed ($V_{\text{tas}}$) with accelerometer drag deceleration to estimate horizontal wind ($\vec{W}$), synthesizing ground track: $\vec{V}_{\text{ground}} = \vec{V}_{\text{air}} + \vec{W}$.
\end{tcolorbox}

\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}EK3\_SRC\_OPTIONS} \hfill \textbf{\color{fmsgray}[SENSOR FALLBACK CASCADE BITMASK]} \\
\textbf{DEFAULT:} \texttt{0} (MANUAL FALLBACK) \hfill \textbf{AUTONOMOUS TARGET:} {\color{fmsgreen}\texttt{1} (AUTO FALLBACK)} \\
\textbf{LSK 5L COMMAND:} {\bfseries\color{fmscyan}param set EK3\_SRC\_OPTIONS 1} \\[0.8mm]
\footnotesize Enables automatic sensor cascade. Setting Bit 0 allows EKF3 to drop an unhealthy GPS autonomously and switch to Dead Reckoning or Optical Flow without pilot intervention.
\end{tcolorbox}

\vspace{0.5mm}

\footnotesize Activates simulated downward optical flow camera and directs EKF3 to ingest pixel displacement rates as primary horizontal velocity ($V_x, V_y$), decoupling hover navigation from satellites.

% SCENARIO 1
\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.6pt,top=1.5mm,bottom=1.5mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}< SCENARIO 1: TOTAL GNSS HARDWARE CUTOFF [CLEAN DEAD RECKONING]}

\vspace{0.5mm}
\textbf{OBJECTIVE:} Evaluate pure aerodynamic and inertial dead reckoning following complete loss of primary satellite reception (antenna severance or hardware failure). Verify aircraft tracks waypoints without wander.

\vspace{0.8mm}
\textbf{PREREQUISITES:} Airspeed calibrated ($\pm 1.0\text{ m/s}$); cruise in \texttt{AUTO} or \texttt{GUIDED} at 50--100 m AGL; drag coefficients set: \texttt{EK3\_DRAG\_BCOEF\_X/Y = 25}.

\vspace{0.8mm}
\textbf{LSK 1L TRIGGER COMMAND:}
\begin{center}
{\bfseries\large\color{fmscyan}param set SIM\_GPS1\_ENABLE 0}
\end{center}

\vspace{0.5mm}
\textbf{TELEMETRY DYNAMICS \& FLIGHT RESPONSE:}
\begin{itemize}[leftmargin=4mm,topsep=1pt,itemsep=0.5pt]
  \item MAVLink \texttt{GPS\_RAW\_INT}: $fix\_type$ drops immediately to \texttt{0} (NO\_FIX), $satellites\_visible = 0$.
  \item EKF3 detects instant satellite loss; cleanly decouples horizontal GPS fusion.
  \item Aircraft enters \textbf{Wind Dead Reckoning}: smoothly banks along waypoint legs, holds track via airspeed and compass heading. Zero weaving or wing-rocking occurs.
\end{itemize}

\vspace{0.8mm}
\textbf{RECOVERY COMMAND:}
{\bfseries\color{fmsgreen}param set SIM\_GPS1\_ENABLE 1} \quad $\rightarrow$ \quad GPS lock reacquired ($fix\_type = 3$); fused in 3--5 s.
\end{tcolorbox}

\vspace{1.5mm}

% SCENARIO 2
\begin{tcolorbox}[colback=fmsboxbg,colframe=fmsborder,arc=1mm,boxrule=0.6pt,top=1.5mm,bottom=1.5mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{fmsdark}< SCENARIO 2: EW RF JAMMING COUNTERMEASURE [ANTI-HUNT PROTOCOL]}

\vspace{0.5mm}
\textbf{OBJECTIVE:} Counter hostile broadband RF noise injection. Prevent aircraft from chasing jumping coordinates (the ``here-and-there'' hunting phenomenon) by forcing automatic innovation rejection and prompt Dead Reckoning fallback.

\vspace{0.8mm}
\textbf{ROOT CAUSE OF WANDERING:}
During RF jamming, receiver does \textit{not} declare $fix\_type = 0$ immediately. It outputs a degraded 3D fix with jumping coordinates. L1 navigation controller turns toward each false jump, and \texttt{AHRS\_GPS\_USE 1} tilts horizon with false ground velocities.

\vspace{0.8mm}
\textbf{STEP 1 --- EXECUTE ANTI-JAM DEFENSIVE PRE-CONFIGURATION (LSK 5L):}
\begin{center}
\begin{tabularx}{\linewidth}{lX}
\toprule
\textbf{COMMAND} & \textbf{OPERATIONAL FUNCTION} \\
\midrule
{\bfseries\color{fmscyan}param set EK3\_SRC\_OPTIONS 1} & Enables EKF3 automatic fallback away from failing GPS. \\
{\bfseries\color{fmscyan}param set FS\_EKF\_THRESH 0.60} & Tightens failsafe threshold to trip instantly under RF noise. \\
{\bfseries\color{fmscyan}param set EK3\_GLITCH\_RAD 15} & Tightens innovation gate from 25 m to 15 m. \\
{\bfseries\color{fmscyan}param set AHRS\_GPS\_USE 0} & Severs GPS velocity from attitude filter to prevent wing-rocking. \\
\bottomrule
\end{tabularx}
\end{center}

\vspace{0.5mm}
\textbf{STEP 2 --- INJECT RF JAMMING NOISE (LSK 2L TRIGGER):}
\begin{center}
{\bfseries\large\color{fmsamber}param set SIM\_GPS1\_JAM 1}
\end{center}

\vspace{0.5mm}
\textbf{TELEMETRY DYNAMICS \& FLIGHT RESPONSE:}
\begin{itemize}[leftmargin=4mm,topsep=1pt,itemsep=0.5pt]
  \item SITL injects carrier noise: receiver HDOP inflates rapidly ($> 4.5$) and position variance surges.
  \item With \texttt{FS\_EKF\_THRESH 0.60}, EKF failsafe trips within 4--6 seconds.
  \item With \texttt{EK3\_SRC\_OPTIONS 1}, EKF3 severs the noisy GPS and cascades into Dead Reckoning.
  \item With \texttt{AHRS\_GPS\_USE 0}, wings remain completely level; erratic weaving is eliminated.
\end{itemize}

\vspace{0.8mm}
\textbf{RECOVERY SEQUENCE:}
{\bfseries\color{fmsgreen}param set SIM\_GPS1\_JAM 0} \quad $\rightarrow$ \quad {\bfseries\color{fmsgreen}param set AHRS\_GPS\_USE 1}
\end{tcolorbox}
