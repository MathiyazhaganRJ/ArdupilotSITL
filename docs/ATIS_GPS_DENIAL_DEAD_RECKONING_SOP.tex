\textbf{\large 1. GNSS Denial, Hardware Cutoff \& Jamming Simulation}

\vspace{1mm}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{SIM\_GPS1\_ENABLE}} \hfill \textbf{Type:} SITL Hardware Toggle \\
\textbf{Default:} \texttt{1} (Active) \quad $\rightarrow$ \quad \textbf{Execution Target:} \texttt{0} (Severed / Denied) \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set SIM\_GPS1\_ENABLE 0} \\[1.5mm]
\textbf{Description:} Simulates complete power cutoff or physical antenna severance to the primary GNSS receiver. When set to \texttt{0}, SITL instantly halts satellite tracking, dropping MAVLink \texttt{GPS\_RAW\_INT} to $fix\_type = 0$ (No GPS) and $satellites\_visible = 0$. This forces the ArduPilot EKF3 core to declare horizontal position invalid and transition immediately into inertial Dead Reckoning mode.
\end{tcolorbox}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{SIM\_GPS1\_JAM}} \hfill \textbf{Type:} SITL Jamming Injection \\
\textbf{Default:} \texttt{0} (Normal RF) \quad $\rightarrow$ \quad \textbf{Execution Target:} \texttt{1} (RF Jammed) \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set SIM\_GPS1\_JAM 1} \\[1.5mm]
\textbf{Description:} Injects broadband RF interference and noise across GPS L1/L2 carrier frequencies. Simulates terrestrial electronic warfare jammers. When set to \texttt{1}, receiver Carrier-to-Noise density ($C/N_0$) drops below the tracking threshold ($< 28\text{ dB-Hz}$), causing carrier phase lock loss, satellite drops, and severe HDOP inflation ($> 5.0$) before fix loss.

\textbf{\large 2. GNSS Spoofing Defense, Innovation Gating \& Attitude Protection}

\vspace{1mm}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{EK3\_GLITCH\_RAD}} \hfill \textbf{Type:} EKF3 Innovation Gate Threshold \\
\textbf{Default:} \texttt{25.0} m \quad $\rightarrow$ \quad \textbf{Recommended Combat Setting:} \texttt{15.0} to \texttt{25.0} m \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set EK3\_GLITCH\_RAD 25} \\[1.5mm]
\textbf{Description:} Defines the maximum horizontal step distance (in meters) allowed between consecutive GPS updates before EKF3 rejects the measurement as a position jump anomaly (glitch or adversarial spoofing attempt). If a spoofed signal attempts to shift the aircraft trajectory faster than natural aerodynamic acceleration limits, EKF3 discards the vector and flags \texttt{GPS\_Glitch}.
\end{tcolorbox}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{FS\_EKF\_THRESH}} \hfill \textbf{Type:} Flight Safety Failsafe Ratio \\
\textbf{Default:} \texttt{0.80} \quad $\rightarrow$ \quad \textbf{Recommended Setting:} \texttt{0.65} to \texttt{0.80} \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set FS\_EKF\_THRESH 0.8} \\[1.5mm]
\textbf{Description:} Normalized innovation variance threshold that triggers the autonomous EKF Failsafe. When sensor measurements consistently conflict with the internal inertial prediction (such as during meaconing or creeping spoofing attacks), exceeding this threshold triggers emergency automated actions (contingency loiter, Dead Reckoning Return-to-Launch, or Land).
\end{tcolorbox}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{AHRS\_GPS\_USE}} \hfill \textbf{Type:} Attitude Filter Sensor Isolation \\

\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set AHRS\_GPS\_USE 0} \\[1.5mm]
\textbf{Description:} Controls whether the primary attitude determination filter uses GPS ground velocity to apply centrifugal compensation during coordinated turns ($a_{\text{cent}} = \omega_z \cdot V_{\text{gps}}$). When GPS spoofing or erratic velocity jumps occur, setting this parameter to \texttt{0} isolates the artificial horizon, preventing erroneous roll/pitch tilt induced by false satellite velocity.
\end{tcolorbox}

% SECTION 3: DEAD RECKONING & AERODYNAMIC WIND ESTIMATION
\textbf{\large 3. Dead Reckoning, True Airspeed \& Aerodynamic Wind Estimation}

\vspace{1mm}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{EK3\_DRAG\_BCOEF\_X}} \hfill \textbf{Type:} Aerodynamic Drag Ballistic Coefficient \\
\textbf{Default:} \texttt{0.0} (Disabled) \quad $\rightarrow$ \quad \textbf{Recommended Airframe Setting:} \texttt{25.0} to \texttt{35.0} ($kg/m^2$) \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set EK3\_DRAG\_BCOEF\_X 25} \\[1.5mm]
\textbf{Description:} Sets the aircraft's longitudinal frontal ballistic drag coefficient for EKF3's wind estimation filter. In GPS-denied forward flight, EKF3 uses this coefficient along with pitot true airspeed ($V_{\text{tas}}$) and accelerometer deceleration to estimate the horizontal wind vector ($\vec{W}_{\text{wind}}$). This bounds inertial dead reckoning drift indefinitely by synthesizing ground velocity: $\vec{V}_{\text{ground}} = \vec{V}_{\text{air}} + \vec{W}_{\text{wind}}$.
\end{tcolorbox}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{EK3\_DRAG\_BCOEF\_Y}} \hfill \textbf{Type:} Lateral Aerodynamic Drag Coefficient \\
\textbf{Default:} \texttt{0.0} (Disabled) \quad $\rightarrow$ \quad \textbf{Recommended Setting:} \texttt{20.0} to \texttt{30.0} ($kg/m^2$) \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set EK3\_DRAG\_BCOEF\_Y 25} \\[1.5mm]
\textbf{Description:} Models lateral fuselage side-force slip dynamics. Enables EKF3 to compute crosswind velocity and crab angle during GPS-denied cruising without requiring dedicated physical side-slip angle vanes.
\end{tcolorbox}

\begin{tcolorbox}[colback=boxbg,colframe=gray!60,arc=1mm,boxrule=0.6pt,top=2mm,bottom=2mm,left=3mm,right=3mm]
\textbf{\large\texttt{EK3\_SRC\_OPTIONS}} \hfill \textbf{Type:} Sensor Fallback Mode Bitmask \\
\textbf{Default:} \texttt{0} (Manual/No Cascade) \quad $\rightarrow$ \quad \textbf{Execution Target:} \texttt{1} (Automatic Fallback) \\
\textbf{MCDU / Console Command:} {\ttfamily\bfseries\color{paramcmd}param set EK3\_SRC\_OPTIONS 1} \\[1.5mm]
\textbf{Description:} Bitmask parameter that enables automatic internal fallback switching. When Bit 0 is set to \texttt{1}, if the primary navigation source (GPS) fails innovation checks or drops fix, EKF3 automatically cascades to secondary available sources (such as Optical Flow or Airspeed/Compass Wind Dead Reckoning) without pilot intervention.
\end{tcolorbox}
