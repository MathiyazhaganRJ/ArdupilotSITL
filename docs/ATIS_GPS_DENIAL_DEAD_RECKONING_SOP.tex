\textbf{1. GNSS HARDWARE CUTOFF \& RF JAMMING SIMULATION}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}SIM\_GPS1\_ENABLE} \hfill \textbf{\small\color{avgray}[HARDWARE TOGGLE]} \\
\textbf{DEFAULT:} \texttt{1} (ACTIVE) \hfill \textbf{TARGET:} {\color{avamber}\texttt{0} (SEVERED / DENIED)} \\
\textbf{LSK 1L COMMAND:} {\bfseries\normalsize\color{avcyan}param set SIM\_GPS1\_ENABLE 0} \\[1mm]
{\normalsize\raggedright Simulates physical antenna severance or receiver power cutoff. SITL instantly sets \texttt{fix\_type = 0} (NO\_FIX) and \texttt{satellites\_visible = 0}. EKF3 immediately declares satellite aiding invalid and transitions directly to smooth Dead Reckoning without coordinate jitter.\par}
\end{tcolorbox}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}SIM\_GPS1\_JAM} \hfill \textbf{\small\color{avgray}[ELECTRONIC WARFARE INJECTION]} \\
\textbf{DEFAULT:} \texttt{0} (NORMAL RF) \hfill \textbf{TARGET:} {\color{avamber}\texttt{1} (RF JAMMED)} \\
\textbf{LSK 2L COMMAND:} {\bfseries\normalsize\color{avcyan}param set SIM\_GPS1\_JAM 1} \\[1mm]
{\normalsize\raggedright Injects broadband RF interference on GPS L1/L2 carrier frequencies (\texttt{C/N0 < 28 dB-Hz}), inflating pseudorange noise. \textbf{CAUTION:} SITL maintains a degraded 3D fix with jumping coordinates; requires automated gating configuration (Section 2) to eliminate aircraft wandering.\par}
\end{tcolorbox}

\vspace{0.5mm}
\textbf{2. INNOVATION GATING \& ATTITUDE HORIZON PROTECTION}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}EK3\_GLITCH\_RAD} \hfill \textbf{\small\color{avgray}[INNOVATION STEP GATE THRESHOLD]} \\
\textbf{DEFAULT:} \texttt{25.0} m \hfill \textbf{RECOMMENDED:} {\color{avgreen}\texttt{15.0} to \texttt{20.0} m} \\
\textbf{MCDU COMMAND:} {\bfseries\normalsize\color{avcyan}param set EK3\_GLITCH\_RAD 15} \\[1mm]
{\normalsize\raggedright Defines maximum allowable horizontal step jump between consecutive GPS updates before EKF3 rejects the measurement as anomalous. Rejects abrupt adversarial spoofing coordinates and extreme jamming spikes.\par}
\end{tcolorbox}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}AHRS\_OPTIONS} \hfill \textbf{\small\color{avgray}[DCM FALLBACK INHIBIT BITMASK]} \\
\textbf{DEFAULT:} \texttt{0} (FALLBACK ALLOWED) \hfill \textbf{ANTI-JAM TARGET:} {\color{avgreen}\texttt{3} (DCM LOCKOUT)} \\
\textbf{MCDU COMMAND:} {\bfseries\normalsize\color{avcyan}param set AHRS\_OPTIONS 3} \\[1mm]
{\normalsize\raggedright Critical anti-hunting parameter. Disables automatic fallback from EKF3 to legacy DCM (\texttt{Bit 0: FW}, \texttt{Bit 1: VTOL}). Under RF jamming, DCM bypasses EKF dead reckoning and feeds raw jumping GPS directly to navigation, causing wild wandering.\par}
\end{tcolorbox}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}AHRS\_GPS\_USE} \hfill \textbf{\small\color{avgray}[ATTITUDE FILTER ISOLATION]} \\
\textbf{DEFAULT:} \texttt{1} (ENABLED) \hfill \textbf{HOSTILE RF TARGET:} {\color{avamber}\texttt{0} (ISOLATE AHRS)} \\
\textbf{MCDU COMMAND:} {\bfseries\normalsize\color{avcyan}param set AHRS\_GPS\_USE 0} \\[1mm]
{\normalsize\raggedright Disconnects GPS ground velocity from attitude centrifugal turn compensation (\texttt{a\_cent = omega * V\_gps}). Prevents erratic GPS velocity spikes from corrupting the artificial horizon and inducing violent wing-rocking.\par}
\end{tcolorbox}

\noindent\textbf{3. DEAD RECKONING \& AERODYNAMIC WIND ESTIMATION}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}EK3\_DRAG\_BCOEF\_X / EK3\_DRAG\_BCOEF\_Y} \hfill \textbf{\small\color{avgray}[DRAG BALLISTICS]} \\
\textbf{DEFAULT:} \texttt{0.0} (DISABLED) \hfill \textbf{AIRFRAME TARGET:} {\color{avgreen}\texttt{25.0} to \texttt{35.0} ($kg/m^2$)} \\
\textbf{LSK 5R COMMANDS:} {\bfseries\normalsize\color{avcyan}param set EK3\_DRAG\_BCOEF\_X 25} \quad|\quad {\bfseries\normalsize\color{avcyan}param set EK3\_DRAG\_BCOEF\_Y 25} \\[1mm]
{\normalsize\raggedright Calibrates longitudinal and lateral airframe ballistic drag. In GPS-denied flight, EKF3 combines pitot true airspeed ($V_{\text{tas}}$) with accelerometer drag deceleration to estimate horizontal wind ($\vec{W}$), synthesizing ground track: $\vec{V}_{\text{ground}} = \vec{V}_{\text{air}} + \vec{W}$.\par}
\end{tcolorbox}

\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.5mm,bottom=1.5mm,left=3mm,right=3mm]
\textbf{\large\color{avdark}EK3\_OPTIONS} \hfill \textbf{\small\color{avgray}[JAMMING EXPECTED EKF BEHAVIOR]} \\
\textbf{DEFAULT:} \texttt{0} (STANDARD EKF) \hfill \textbf{ANTI-JAM TARGET:} {\color{avgreen}\texttt{1} (JAMMING EXPECTED)} \\
\textbf{LSK 4L COMMAND:} {\bfseries\normalsize\color{avcyan}param set EK3\_OPTIONS 1} \\[1mm]
{\normalsize\raggedright Bit 0 (\texttt{JammingExpected}). In EKF3, prevents automatic position reset to degraded GPS upon timeout. Sets \texttt{waitingForGpsChecks = true}, isolating the filter in pure Dead Reckoning until all pre-flight quality checks pass continuously.\par}
\end{tcolorbox}

% SCENARIO 1

\textbf{\large\color{avdark}SCENARIO 1: TOTAL GNSS HARDWARE CUTOFF [CLEAN DEAD RECKONING]}

\vspace{0.3mm}
{\normalsize\raggedright\textbf{OBJECTIVE:} Evaluate pure aerodynamic dead reckoning upon complete loss of primary satellite reception (antenna severance / power loss). Verify track hold without wander.\par}

\vspace{0.3mm}
{\normalsize\textbf{PREREQUISITES:} Airspeed valid; cruise in \texttt{AUTO} at 50--100 m AGL; drag set: \texttt{EK3\_DRAG\_BCOEF\_X/Y = 25}.\par}

\vspace{0.4mm}
\textbf{LSK 1L TRIGGER:} {\bfseries\large\color{avcyan}param set SIM\_GPS1\_ENABLE 0}

\vspace{0.3mm}

\begin{itemize}[leftmargin=4mm,topsep=1pt,itemsep=0.5pt]
  \item {\normalsize\raggedright \texttt{fix\_type} drops instantly to \texttt{0} (NO\_FIX), \texttt{satellites\_visible = 0}.\par}
  \item {\normalsize\raggedright EKF3 cleanly decouples GPS horizontal fusion; transitions immediately to Dead Reckoning.\par}
  \item {\normalsize\raggedright Aircraft holds track via airspeed and compass heading; zero wandering or wing-rock.\par}
\end{itemize}

\vspace{0.2mm}
{\normalsize\textbf{RECOVERY:} {\bfseries\color{avgreen}param set SIM\_GPS1\_ENABLE 1} ($\rightarrow$ GPS lock restored in 3--5 s).\par}

% SCENARIO 2
\begin{tcolorbox}[colback=avboxbg,colframe=avborder,arc=1mm,boxrule=0.5pt,top=1.2mm,bottom=1.2mm,left=2.5mm,right=2.5mm]
\textbf{\large\color{avdark}SCENARIO 2: EW RF JAMMING COUNTERMEASURE [ANTI-HUNT PROTOCOL]}

\vspace{0.3mm}
{\normalsize\raggedright\textbf{OBJECTIVE:} Counter broadband RF jamming. Prevent coordinate chasing (``here-and-there'' hunting) by locking out legacy DCM fallback and locking EKF3 in aerodynamic Dead Reckoning.\par}

\vspace{0.3mm}
{\normalsize\raggedright\textbf{ROOT CAUSE OF WANDERING:} (1) Default \texttt{AHRS\_OPTIONS = 0} falls back to DCM when EKF rejects GPS but \texttt{status >= 3D}; DCM has \textit{no} dead reckoning and feeds raw jumping GPS to navigation. (2) Without \texttt{EK3\_OPTIONS = 1}, EKF resets position to bad GPS on timeout. (3) Dead reckoning requires a converged wind vector: $\vec{V}_{\text{gnd}} = \vec{V}_{\text{tas}} + \vec{W}$.\par}

\vspace{0.3mm}
\textbf{STEP 1 --- PRE-FLIGHT ANTI-JAM SETUP (LSK 4L, 4R, 5L, 5R):}\\
{\bfseries\color{avcyan}param set AHRS\_OPTIONS 3} \quad [Inhibits DCM fallback; stays in EKF3]\\
{\bfseries\color{avcyan}param set EK3\_OPTIONS 1} \quad [JammingExpected; blocks bad position reset]\\
{\bfseries\color{avcyan}param set AHRS\_GPS\_USE 0} \quad [Severs GPS velocity from attitude horizon]\\
{\bfseries\color{avcyan}param set EK3\_DRAG\_BCOEF\_X 25} \quad [Ballistic drag for wind vector solution]

\vspace{0.3mm}
\textbf{STEP 2 --- MANDATORY WIND-LEARNING ORBIT (PRIOR TO JAMMING):}\\
{\normalsize\raggedright Fly \texttt{AUTO} or \texttt{LOITER} turns for \textbf{20 seconds} with valid GPS to converge wind states ($\vec{W}$).\par}

\vspace{0.3mm}
\textbf{STEP 3 --- INJECT RF JAMMING (LSK 2L TRIGGER):} {\bfseries\large\color{avamber}param set SIM\_GPS1\_JAM 1}

\vspace{0.3mm}
\textbf{TELEMETRY DYNAMICS \& FLIGHT RESPONSE:}
\begin{itemize}[leftmargin=4mm,topsep=1pt,itemsep=0.5pt]
  \item {\normalsize\raggedright SITL injects noise; EKF3 detects innovation errors and isolates GPS inputs.\par}
  \item {\normalsize\raggedright \texttt{AHRS\_OPTIONS 3} blocks DCM fallback; navigation stays strictly in EKF3.\par}
  \item {\normalsize\raggedright \texttt{EK3\_OPTIONS 1} holds Dead Reckoning without resetting position to noise.\par}
  \item {\normalsize\raggedright Aircraft tracks waypoints accurately; wandering and hunting are eliminated.\par}
\end{itemize}

\vspace{0.2mm}
{\normalsize\textbf{RECOVERY:} {\bfseries\color{avgreen}param set SIM\_GPS1\_JAM 0} \quad $\rightarrow$ \quad {\bfseries\color{avgreen}param set AHRS\_GPS\_USE 1}\par}
\end{tcolorbox}
